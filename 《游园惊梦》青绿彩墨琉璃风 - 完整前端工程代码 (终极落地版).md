3. 主控路由枢纽：泼墨转场引擎 (src/App.jsx)
这是加入“水墨国画风格衔接”的核心。不再是死板的切换，而是每一次切换都伴随一次绝美的墨水晕染。

JavaScript
import React, { useState, useRef } from 'react';
import { gsap } from 'gsap';
import Scene1_Fog from './components/Scene1_Fog';
import Scene2_Model from './components/Scene2_Model';
import Scene3_Light from './components/Scene3_Light';
import Scene4_Parallax from './components/Scene4_Parallax';

export default function App() {
  const [currentScene, setCurrentScene] = useState(1);
  const [isTransitioning, setIsTransitioning] = useState(false);
  const inkOverlayRef = useRef(null);

  // 核心前端动画：泼墨转场引擎
  const handleSceneChange = (nextSceneIndex) => {
    if (isTransitioning) return;
    setIsTransitioning(true);

    // 第一阶段：墨水从屏幕中央疯狂晕开，遮盖视野
    gsap.fromTo(inkOverlayRef.current, 
      { clipPath: 'circle(0% at 50% 50%)', opacity: 1 },
      { 
        clipPath: 'circle(150% at 50% 50%)', 
        duration: 1.8, 
        ease: 'power2.inOut',
        onComplete: () => {
          // 视野完全被墨水遮挡时，瞬间切换背后的组件
          setCurrentScene(nextSceneIndex);
          
          // 第二阶段：墨水渐渐淡出/散去，露出新场景
          gsap.to(inkOverlayRef.current, { 
            opacity: 0, 
            duration: 1.5, 
            ease: 'power2.inOut', 
            delay: 0.2, // 稍微停顿一下，增加意境
            onComplete: () => setIsTransitioning(false)
          });
        }
      }
    );
  };

  return (
    <div className="w-screen h-screen overflow-hidden bg-black text-white relative">
      
      {/* --- 全局 SVG 滤镜库 --- */}
      <svg className="hidden">
        {/* 用于文字的轻度水墨滤镜 */}
        <filter id="ink-bleed-text">
          <feTurbulence type="fractalNoise" baseFrequency="0.04" numOctaves="3" result="noise" />
          <feDisplacementMap in="SourceGraphic" in2="noise" scale="15" xChannelSelector="R" yChannelSelector="G" />
          <feGaussianBlur stdDeviation="2" result="blur" />
          <feMerge>
            <feMergeNode in="blur" />
            <feMergeNode in="SourceGraphic" />
          </feMerge>
        </filter>
        
        {/* 用于转场画面的重度水墨湍流滤镜 */}
        <filter id="ink-transition-splash">
          <feTurbulence type="fractalNoise" baseFrequency="0.015" numOctaves="4" result="warp" />
          <feDisplacementMap in="SourceGraphic" in2="warp" scale="60" xChannelSelector="R" yChannelSelector="G" />
        </filter>
      </svg>

      {/* --- 泼墨转场遮罩层 --- */}
      <div 
        ref={inkOverlayRef} 
        className="absolute inset-0 z-50 pointer-events-none bg-[#0a0c0a]"
        style={{ filter: 'url(#ink-transition-splash)', opacity: 0 }}
      ></div>

      {/* --- 场景挂载区 --- */}
      {currentScene === 1 && <Scene1_Fog onComplete={() => handleSceneChange(2)} />}
      {currentScene === 2 && <Scene2_Model onNext={() => handleSceneChange(3)} />}
      {currentScene === 3 && <Scene3_Light onNext={() => handleSceneChange(4)} />}
      {currentScene === 4 && <Scene4_Parallax />}
      
    </div>
  );
}

import React, { useRef, useEffect, useState } from 'react';
import { gsap } from 'gsap';

export default function Scene1_Fog({ onComplete }) {
  const canvasRef = useRef(null);
  const [isDrawing, setIsDrawing] = useState(false);
  const [scratchedArea, setScratchedArea] = useState(0);

  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas.getContext('2d');
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;

    // 绘制低饱和度青绿水汽层
    ctx.fillStyle = 'rgba(136, 171, 138, 0.98)'; 
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    const handleScratch = (e) => {
      if (!isDrawing) return;
      const rect = canvas.getBoundingClientRect();
      const x = (e.clientX || e.touches?.[0].clientX) - rect.left;
      const y = (e.clientY || e.touches?.[0].clientY) - rect.top;

      ctx.globalCompositeOperation = 'destination-out';
      ctx.beginPath();
      ctx.arc(x, y, 90, 0, Math.PI * 2);
      ctx.filter = 'blur(15px)'; // 彩墨边缘晕开效果
      ctx.fill();

      setScratchedArea(prev => prev + 1);
      // 当擦除面积达标后，触发 onComplete，交由 App.jsx 执行泼墨转场
      if (scratchedArea > 120) { 
        setIsDrawing(false);
        onComplete();
      }
    };

    canvas.addEventListener('mousedown', () => setIsDrawing(true));
    canvas.addEventListener('mouseup', () => setIsDrawing(false));
    canvas.addEventListener('mousemove', handleScratch);
    canvas.addEventListener('touchstart', () => setIsDrawing(true));
    canvas.addEventListener('touchend', () => setIsDrawing(false));
    canvas.addEventListener('touchmove', handleScratch);

    return () => {
      canvas.removeEventListener('mousedown', () => setIsDrawing(true));
      canvas.removeEventListener('mouseup', () => setIsDrawing(false));
      canvas.removeEventListener('mousemove', handleScratch);
    };
  }, [isDrawing, scratchedArea, onComplete]);

  return (
    <div className="absolute inset-0 w-full h-full">
      <img src="/bg-clear-garden.jpg" className="absolute inset-0 w-full h-full object-cover z-0" alt="江南春色" />
      <canvas ref={canvasRef} className="absolute inset-0 z-10 cursor-pointer" />
      <div className="absolute inset-0 z-20 pointer-events-none flex items-center justify-center">
        <div className="glass-panel text-white px-10 py-5 rounded-full text-lg tracking-widest animate-pulse">
          擦拭晨雾，得见江南
        </div>
      </div>
    </div>
  );
}

import React, { useState, useRef } from 'react';
import { gsap } from 'gsap';

const POEMS = [
  "不到园林，怎知春色如许。",
  "清风明月本无价，近水远山皆有情。",
  "疏影横斜水清浅，暗香浮动月黄昏。"
];

export default function Scene3_Light({ onNext }) {
  const colorBgRef = useRef(null);
  const [isLit, setIsLit] = useState(false);
  const [currentPoem, setCurrentPoem] = useState("");

  const handleLightUp = (e) => {
    if (isLit) return;
    setIsLit(true);
    
    setCurrentPoem(POEMS[Math.floor(Math.random() * POEMS.length)]);

    const rect = e.target.getBoundingClientRect();
    const x = rect.left + rect.width / 2;
    const y = rect.top + rect.height / 2;

    // 前端动画：彩墨炸开特效
    gsap.fromTo(colorBgRef.current, 
      { clipPath: `circle(0px at ${x}px ${y}px)` },
      { clipPath: `circle(2500px at ${x}px ${y}px)`, duration: 3, ease: "power2.inOut" }
    );
  };

  return (
    <div className="absolute inset-0 w-full h-full bg-black">
      <img src="/ink-dark-bg.jpg" className="absolute inset-0 w-full h-full object-cover z-0 opacity-80" alt="Dark Ink" />
      <img ref={colorBgRef} src="/ink-color-bg.jpg" className="absolute inset-0 w-full h-full object-cover z-10" style={{ clipPath: 'circle(0px at 50% 50%)' }} alt="Color Night" />

      {/* 交互灯笼 */}
      <div className="absolute z-20 top-[40%] left-1/2 -translate-x-1/2 cursor-pointer animate-pulse" onClick={handleLightUp}>
        <img src="/lantern-icon.png" alt="点灯" className={`w-20 h-20 transition-all duration-1000 ${isLit ? 'brightness-[2.0] drop-shadow-[0_0_40px_rgba(255,150,0,1)] scale-110' : 'brightness-50'}`} />
      </div>

      {/* 诗词水墨浮现区 */}
      <div className="absolute bottom-24 right-32 z-30 pointer-events-none">
        <p className={`text-[#FDF6E3] text-4xl font-serif tracking-[0.3em] ink-text ${isLit ? 'visible' : ''}`} style={{ writingMode: 'vertical-rl', textShadow: '2px 2px 8px rgba(0,0,0,0.5)' }}>
          {currentPoem}
        </p>
      </div>

      {isLit && (
        <button onClick={onNext} className="absolute bottom-12 left-1/2 -translate-x-1/2 z-30 glass-panel px-8 py-3 text-white rounded-full hover:bg-white/20 transition delay-1000 animate-fade-in">
          定制专属窗景
        </button>
      )}
    </div>
  );
}

import React, { useRef, useEffect } from 'react';
import { gsap } from 'gsap';

export default function Scene4_Parallax() {
  const rockRef = useRef(null); 
  const pavilionRef = useRef(null); 
  const floraRef = useRef(null);

  useEffect(() => {
    const handleMouseMove = (e) => {
      const { innerWidth, innerHeight } = window;
      const x = (e.clientX / innerWidth - 0.5) * 2; 
      const y = (e.clientY / innerHeight - 0.5) * 2;

      // 前端动画：深度裸眼 3D 视差计算
      gsap.to(rockRef.current, { x: x * 50, y: y * 50, duration: 1.2, ease: "power2.out" });
      gsap.to(pavilionRef.current, { x: x * 20, y: y * 20, duration: 1.2, ease: "power2.out" });
      gsap.to(floraRef.current, { x: x * -15, y: y * -15, duration: 1.2, ease: "power2.out" });
    };

    window.addEventListener('mousemove', handleMouseMove);
    return () => window.removeEventListener('mousemove', handleMouseMove);
  }, []);

  return (
    <div className="absolute inset-0 w-full h-full bg-[#1a1f1b]">
      
      {/* 视差层 */}
      <div className="absolute inset-0 z-0 flex items-center justify-center pointer-events-none">
         <img ref={floraRef} src="/element-flora.png" className="absolute top-10 right-10 w-[40%] opacity-80" alt="Flora" />
         <img ref={pavilionRef} src="/element-pavilion.png" className="absolute w-[55%] opacity-95 drop-shadow-2xl" alt="Pavilion" />
         <img ref={rockRef} src="/element-taihu-rock.png" className="absolute bottom-[-5%] left-[5%] w-[45%] drop-shadow-[0_20px_30px_rgba(0,0,0,0.8)] z-10" alt="Rock" />
